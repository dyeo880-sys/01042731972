// ============================================================
// [최종 에러 수정본] 캡핑기 / 용기 고정 릴레이 제어 - 아두이노 우노
// ============================================================

// ---- 출력 핀 (릴레이/솔밸브 등) ----
const int PIN_CAPPING     = 3;  // 1번 릴레이: 캡핑기 하강 솔밸브
const int PIN_TIMER1      = 4;  // 2번 릴레이: 1번 외장 타이머 기동
const int PIN_TIMER2      = 5;  // 3번 릴레이: 2번 외장 타이머 (캡 비비기 회전)
const int PIN_FIX_CYL     = 6;  // 4번 릴레이: 용기 고정 실린더 솔밸브
const int PIN_LED         = 13; // 보드 내부 LED

// ---- 입력 핀 (센서/버튼) ----
const int PIN_CONVEYOR_PHOTO = 7;  // IN1: 컨베이어 24V 광센서
const int PIN_TIMER1_DONE    = 8;  // IN2: 1번 외장 타이머 완료 신호선
const int PIN_SENSOR_PRESS   = 9;  // IN3: 용기 누름 센서 가동 신호선
const int PIN_SENSOR_LOWER   = 10; // IN4: 캡핑기 하단 센서 / 수동버튼1
const int PIN_SENSOR_TOP     = 11; // IN5: 캡핑기 상단 센서 / 수동버튼2
const int PIN_BTN_MODE4      = 12; // IN6: 수동 3번 버튼 (연동 기동 및 자동 복귀)

int testStatus = 0; // 0 = 정상 자동 모드, 1 = 테스트(수동) 모드

void setup() {
  // 출력 핀 설정
  pinMode(PIN_CAPPING, OUTPUT);
  pinMode(PIN_TIMER1, OUTPUT);
  pinMode(PIN_TIMER2, OUTPUT);
  pinMode(PIN_FIX_CYL, OUTPUT);
  pinMode(PIN_LED, OUTPUT);

  // 입력 핀 설정 (DNUNA16 보드 내부 절연 풀업 구조 반영)
  pinMode(PIN_CONVEYOR_PHOTO, INPUT_PULLUP);
  pinMode(PIN_TIMER1_DONE, INPUT_PULLUP);
  pinMode(PIN_SENSOR_PRESS, INPUT_PULLUP);
  pinMode(PIN_SENSOR_LOWER, INPUT_PULLUP);
  pinMode(PIN_SENSOR_TOP, INPUT_PULLUP);
  pinMode(PIN_BTN_MODE4, INPUT_PULLUP);

  // ── 전원 ON 사고 방지 및 최초 안전 차단 ──
  digitalWrite(PIN_CAPPING, HIGH); // 차단/대기
  digitalWrite(PIN_TIMER1, HIGH);  // 차단/대기
  digitalWrite(PIN_TIMER2, HIGH);  // 차단/대기
  digitalWrite(PIN_FIX_CYL, HIGH); // 차단/대기
  digitalWrite(PIN_LED, LOW);      // OFF

  testStatus = 0; // 전원 켜지면 무조건 자동 모드로 시작

  // 🔥 [돌발 사고 방지] 전원을 켤 때 센서가 미리 눌려있어도 3초간 기계를 완전히 얼려둡니다.
  delay(3000);
}

void loop() {
  // ━━━━━━ 모드 4 : 수동 3번 버튼 연동 기동 및 정상 자동 가동 복귀 ━━━━━━
  if (digitalRead(PIN_BTN_MODE4) == LOW) {

    // 1단계: 용기 누름 센서 신호 대기
    while (digitalRead(PIN_SENSOR_PRESS) != LOW) { /* 대기 */ }

    // 2단계: 용기 내림 위치 센서 대기
    while (digitalRead(PIN_SENSOR_LOWER) != LOW) { /* 대기 */ }

    // 3단계: 확인 완료 후 비비기 타이머 가동
    digitalWrite(PIN_TIMER2, LOW);   // 2번 타이머 ON
    delay(3000);                     // 3초간 비비기 작업 대기
    digitalWrite(PIN_TIMER2, HIGH);  // 2번 타이머 OFF

    // 4단계: 작업 종료 후 캡핑기 상승 복귀
    digitalWrite(PIN_CAPPING, HIGH); // 캡핑기 하강 해제 -> 상승

    // 5단계: 완전히 복귀한 상단 신호 대기
    while (digitalRead(PIN_SENSOR_TOP) != LOW) { /* 대기 */ }

    // 6단계: 상단 도달 후 마무리 타이머 (1초 지연)
    delay(1000);

    // 7단계: 꽉 잡고 있던 실린더 풀어줌
    digitalWrite(PIN_FIX_CYL, HIGH); // 용기 고정 해제

    // 8단계: 수동 종료 완료! 다시 컨베이어 광센서 작동 부활 (자동 복귀)
    testStatus = 0;

  } else {

    // ━━━━━━ 모드 1 : 컨베이어 정상 자동 가동 (14단계 마스터 공정) ━━━━━━
    if (testStatus == 0 && digitalRead(PIN_CONVEYOR_PHOTO) == LOW) {
      
      // [자동 1] 1번 타이머 작동
      digitalWrite(PIN_TIMER1, LOW);
      // [자동 2] 1번 타이머 완료 신호(8번 핀) 대기
      while (digitalRead(PIN_TIMER1_DONE) != LOW) { /* 대기 */ }
      // [자동 3] 1번 타이머 꺼짐
      digitalWrite(PIN_TIMER1, HIGH);
      
      // [자동 4] 용기 고정 실린더 작동
      digitalWrite(PIN_FIX_CYL, LOW);
      // [자동 5] 실린더 고정 완료 센서(9번 핀) 대기
      while (digitalRead(PIN_SENSOR_PRESS) != LOW) { /* 대기 */ }
      
      // [자동 6] 캡핑기 하강 시작
      digitalWrite(PIN_CAPPING, LOW);
      // [자동 7] 내림 위치 센서(10번 핀) 대기
      while (digitalRead(PIN_SENSOR_LOWER) != LOW) { /* 대기 */ }
      
      // [자동 8] 2번 타이머(비비기) 작동
      digitalWrite(PIN_TIMER2, LOW);
      // [자동 9] 캡 비비는 시간 3초 대기
      delay(3000);
      // [자동 10] 2번 타이머 꺼짐
      digitalWrite(PIN_TIMER2, HIGH);
      
      // [자동 11] 캡핑기 상승 복귀
      digitalWrite(PIN_CAPPING, HIGH);
      // [자동 12] 올림 위치 센서(11번 핀) 대기
      while (digitalRead(PIN_SENSOR_TOP) != LOW) { /* 대기 */ }
      
      // [자동 13] 용기 고정 실린더 풀어줌
      digitalWrite(PIN_FIX_CYL, HIGH);
      // [자동 14] 실린더 완전히 빠질 때까지 대기 후 완료 리셋
      delay(500);
    }

    // ━━━━━━ 모드 2 : 수동 버튼 1번 테스트 (캡핑기 하강) ━━━━━━
    if (digitalRead(PIN_SENSOR_LOWER) == LOW) {
      testStatus = 1;                  // 테스트 상태 진입 (자동 광센서 중지)
      digitalWrite(PIN_CAPPING, LOW);  // 1번 버튼 작동: 캡핑기 내려옴
    }

    // ━━━━━━ 모드 3 : 수동 버튼 2번 테스트 (용기 고정) ━━━━━━
    if (digitalRead(PIN_SENSOR_TOP) == LOW) {
      testStatus = 1;                  // 테스트 상태 진입 (자동 광센서 중지)
      digitalWrite(PIN_FIX_CYL, LOW);  // 2번 버튼 작동: 용기 잡는 실린더 작동
    }
  }
}
